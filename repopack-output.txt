function slug(value: string) {
  return value
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .toUpperCase()
    .replace(/[^A-Z0-9]+/g, "-")
    .replace(/^-|-$/g, "");
}

function sizeCode(size: string) {
  return size.replace("XS/S", "XSS").replace("S/M", "SM").replace("/", "");
}

function makeSku(name: string, color: string, size: string) {
  return `${slug(name)}-${slug(color)}-${sizeCode(size)}`;
}

function normalizeProduct(raw: any): Product {
  const dollarPrice = Number(
    raw.dollarPrice ?? raw.dollar_price ?? raw.price_usd ?? 0,
  );

  if (Array.isArray(raw.variants)) {
    return {
      id: String(raw.id),
      name: raw.name || "Sin nombre",
      category: raw.category || "Tops",

      dollarPrice,

      price: Number(raw.price || 0),
      cost: Number(raw.cost || raw.cost_usd || 0),

      webVisible: raw.webVisible !== false,

      images: raw.images || [],
      sizeGuideImages: raw.sizeGuideImages || raw.size_guide_images || [],

      careDetails: raw.careDetails || raw.care || "",

      variants: raw.variants.map((v: any) => ({
        id: String(v.id || Date.now()),
        color: v.color || "Único",
        size: v.size || "U",

        sku:
          v.sku ||
          makeSku(raw.name || "PRODUCTO", v.color || "UNICO", v.size || "U"),

        initialStock: Number(v.initialStock ?? v.initial_stock ?? v.stock ?? 0),

        stock: Number(v.stock ?? 0),
      })),
    };
  }

  // Compatibilidad con estructura anterior
  return {
    id: String(raw.id),
    name: raw.name || "Sin nombre",
    category: raw.category || "Tops",

    dollarPrice,

    price: Number(raw.price || 0),
    cost: Number(raw.cost || raw.cost_usd || 0),

    webVisible: raw.webVisible !== false,

    images: raw.images || [],
    sizeGuideImages: raw.sizeGuideImages || raw.size_guide_images || [],

    careDetails: raw.careDetails || raw.care || "",

    variants: [
      {
        id: `legacy-${raw.id}`,
        color: raw.color || "Único",
        size: raw.size || "U",

        sku:
          raw.sku ||
          makeSku(
            raw.name || "PRODUCTO",
            raw.color || "UNICO",
            raw.size || "U",
          ),

        initialStock: Number(
          raw.initialStock ?? raw.initial_stock ?? raw.stock ?? 0,
        ),

        stock: Number(raw.stock ?? 0),
      },
    ],
  };
}

// Función auxiliar: Buscar o crear categoría por nombre
async function getOrCreateCategory(categoryName: string) {
  if (!categoryName) return null;
  const normalizedName = categoryName.trim();

  // 1. Buscar si ya existe
  const { data: existingCategory, error: fetchError } = await supabase
    .from("categories")
    .select("id")
    .ilike("name", normalizedName)
    .maybeSingle();

  if (fetchError) throw fetchError;
  if (existingCategory) return existingCategory.id;

  // 2. Crear si no existe
  const { data: newCategory, error: insertError } = await supabase
    .from("categories")
    .insert({ name: normalizedName })
    .select("id")
    .single();

  if (insertError) throw insertError;
  return newCategory.id;
}

export async function GET() {

// POST: Crear Producto, Variantes e Historial Inicial
// ========================================================
export async function POST(req: Request) {
  let createdProductId: string | null = null;

  try {
    const body = await req.json();
    const {
      name,
      description,
      dollar_price,
      category_name,
      images,
      show_web,
      variants,
    } = body;

    if (!name || dollar_price == null) {
      return NextResponse.json(
        { error: "El nombre y el precio en dólares son requeridos" },
        { status: 400 },
      );
    }

    // 1. Obtener o crear ID de categoría
    const categoryId = category_name
      ? await getOrCreateCategory(category_name)
      : null;

    // 2. Insertar registro principal del Producto
    const { data: newProduct, error: productError } = await supabase
      .from("products")
      .insert({
        name,
        description: description || "",
        dollar_price: Number(dollar_price),
        category_id: categoryId,
        images: images || [],
        show_web: show_web ?? true,
      })
      .select("id")
      .single();

    if (productError || !newProduct)
      throw productError || new Error("Error al crear el producto");
    createdProductId = newProduct.id;

    // 3. Insertar Variantes asociadas
    let createdVariants: any[] = [];
    if (Array.isArray(variants) && variants.length > 0) {
      const variantsToInsert = variants.map((v: any) => ({
        product_id: createdProductId,
        color: v.color || null,
        size: v.size || null,
        stock: Number(v.stock || 0),
        sku: v.sku || null,
      }));

      const { data: insertedVariants, error: variantsError } = await supabase
        .from("product_variants")
        .insert(variantsToInsert)
        .select();

      if (variantsError) throw variantsError;
      createdVariants = insertedVariants || [];
    }

    // 4. Registrar en Historial de Stock si hay cantidades iniciales mayores a 0
    const historyEntries = createdVariants
      .filter((v: any) => Number(v.stock) > 0)
      .map((v: any) => ({
        product_id: createdProductId,
        variant_id: v.id,
        quantity_change: Number(v.stock),
        type: "INITIAL_STOCK",
        notes: "Stock inicial registrado al crear producto",
      }));

    if (historyEntries.length > 0) {
      const { error: historyError } = await supabase
        .from("stock_history")
        .insert(historyEntries);
      if (historyError)
        console.error("Error registrando historial inicial:", historyError);
    }

    return NextResponse.json(
      { success: true, productId: createdProductId },
      { status: 201 },
    );
  } catch (error: any) {
    console.error("Error en POST /api/products:", error);

    // Rollback manual: Si se creó el producto pero fallaron las variantes o procesos posteriores
    if (createdProductId) {
      await supabase.from("products").delete().eq("id", createdProductId);
    }

    return NextResponse.json(
      { error: error.message || "Error al guardar el producto" },
      { status: 500 },
    );
  }
}

// PATCH: Modificaciones (Visibilidad, Ajuste Stock, Edición)
// ========================================================
export async function PATCH(req: Request) {
  try {
    const body = await req.json();
    const { action } = body;

    switch (action) {
      case "toggle-visibility": {
        const { productId, show_web } = body;
        if (!productId)
          return NextResponse.json(
            { error: "productId es requerido" },
            { status: 400 },
          );

        const { error } = await supabase
          .from("products")
          .update({ show_web: Boolean(show_web) })
          .eq("id", productId);

        if (error) throw error;
        return NextResponse.json({ success: true });
      }

      case "update-stock": {
        const { variantId, change, reason } = body;
        if (!variantId || change == null)
          return NextResponse.json(
            { error: "variantId y change son requeridos" },
            { status: 400 },
          );

        // Obtener stock actual
        const { data: variant, error: fetchErr } = await supabase
          .from("product_variants")
          .select("id, product_id, stock")
          .eq("id", variantId)
          .single();

        if (fetchErr || !variant)
          throw fetchErr || new Error("Variante no encontrada");

        const newStock = Math.max(
          0,
          Number(variant.stock || 0) + Number(change),
        );

        // Actualizar stock de la variante
        const { error: updateErr } = await supabase
          .from("product_variants")
          .update({ stock: newStock })
          .eq("id", variantId);

        if (updateErr) throw updateErr;

        // Registrar entrada/salida en el historial
        if (Number(change) !== 0) {
          await supabase.from("stock_history").insert({
            product_id: variant.product_id,
            variant_id: variantId,
            quantity_change: Number(change),
            type: Number(change) > 0 ? "RESTOCK" : "ADJUSTMENT",
            notes:
              reason ||
              (Number(change) > 0
                ? "Reabastecimiento de stock"
                : "Ajuste manual de stock"),
          });
        }

        return NextResponse.json({ success: true, newStock });
      }

      case "update-product": {
        const {
          productId,
          name,
          description,
          dollar_price,
          category_name,
          images,
          show_web,
          variants,
        } = body;
        if (!productId)
          return NextResponse.json(
            { error: "productId es requerido" },
            { status: 400 },
          );

        // 1. Obtener o crear ID de categoría
        const categoryId = category_name
          ? await getOrCreateCategory(category_name)
          : null;

        // 2. Actualizar datos principales del producto
        const { error: prodUpdateErr } = await supabase
          .from("products")
          .update({
            name,
            description,
            dollar_price: Number(dollar_price),
            category_id: categoryId,
            images: images || [],
            show_web: Boolean(show_web),
            updated_at: new Date().toISOString(),
          })
          .eq("id", productId);

        if (prodUpdateErr) throw prodUpdateErr;

        // 3. Gestionar actualización de variantes
        if (Array.isArray(variants)) {
          // Consultar variantes existentes en la base de datos
          const { data: existingVariants, error: fetchVarErr } = await supabase
            .from("product_variants")
            .select("id, stock")
            .eq("product_id", productId);

          if (fetchVarErr) throw fetchVarErr;

          const existingMap = new Map(
            (existingVariants || []).map((v) => [v.id, Number(v.stock || 0)]),
          );
          const incomingIds = new Set(
            variants.filter((v: any) => v.id).map((v: any) => v.id),
          );

          // Eliminar variantes que se retiraron en el cliente
          const idsToDelete = (existingVariants || [])
            .filter((v) => !incomingIds.has(v.id))
            .map((v) => v.id);

          if (idsToDelete.length > 0) {
            const { error: delErr } = await supabase
              .from("product_variants")
              .delete()
              .in("id", idsToDelete);
            if (delErr) throw delErr;
          }

          // Crear o actualizar variantes mediante upsert
          const historyEntries: any[] = [];

          for (const v of variants) {
            const variantPayload = {
              ...(v.id ? { id: v.id } : {}),
              product_id: productId,
              color: v.color || null,
              size: v.size || null,
              stock: Number(v.stock || 0),
              sku: v.sku || null,
            };

            const { data: savedVariant, error: upsertErr } = await supabase
              .from("product_variants")
              .upsert(variantPayload)
              .select("id, stock")
              .single();

            if (upsertErr) throw upsertErr;

            // Calcular diferencia de stock para registrar solo incrementos positivos
            const oldStock =
              v.id && existingMap.has(v.id) ? existingMap.get(v.id)! : 0;
            const newStock = Number(v.stock || 0);
            const stockIncrease = newStock - oldStock;

            if (stockIncrease > 0) {
              historyEntries.push({
                product_id: productId,
                variant_id: savedVariant.id,
                quantity_change: stockIncrease,
                type: v.id ? "RESTOCK" : "INITIAL_STOCK",
                notes: v.id
                  ? "Incremento de stock por edición"
                  : "Stock inicial de nueva variante",
              });
            }
          }

          if (historyEntries.length > 0) {
            await supabase.from("stock_history").insert(historyEntries);
          }
        }

        return NextResponse.json({ success: true });
      }

      default:
        return NextResponse.json(
          { error: "Acción no válida" },
          { status: 400 },
        );
    }
  } catch (error: any) {
    console.error("Error en PATCH /api/products:", error);
    return NextResponse.json(
      { error: error.message || "Error al actualizar el producto" },
      { status: 500 },
    );
  }
}

export async function POST(request: Request) {
  const supabase = await createClient();

  const body = await request.json();

  const tasaBnc = Number(body.tasaBnc);
  const tasaBcv = Number(body.tasaBcv);

  if (!Number.isFinite(tasaBnc) || !Number.isFinite(tasaBcv)) {
    return NextResponse.json(
      { error: "Las tasas deben ser números válidos." },
      { status: 400 },
    );
  }

  const { data: existingSettings } = await supabase
    .from("store_settings")
    .select("id")
    .limit(1)
    .maybeSingle();

  let data;
  let error;

  if (existingSettings) {
    const result = await supabase
      .from("store_settings")
      .update({
        tasa_bnc: tasaBnc,
        tasa_bcv: tasaBcv,
        updated_at: new Date().toISOString(),
      })
      .eq("id", existingSettings.id)
      .select("id, tasa_bnc, tasa_bcv")
      .single();

    data = result.data;
    error = result.error;
  } else {
    const result = await supabase
      .from("store_settings")
      .insert({
        tasa_bnc: tasaBnc,
        tasa_bcv: tasaBcv,
      })
      .select("id, tasa_bnc, tasa_bcv")
      .single();

    data = result.data;
    error = result.error;
  }

  if (error) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }

  return NextResponse.json({
    settings: data,
  });
}

// Helpers para normalización y parseo
function safeDecode(str: string): string {
  if (!str) return "";
  let decoded = str;
  try {
    for (let i = 0; i < 3; i++) {
      if (decoded.includes("%")) {
        const temp = decodeURIComponent(decoded);
        if (temp === decoded) break;
        decoded = temp;
      }
    }
  } catch (e) {
    console.error("Error decodificando:", e);
  }
  return decoded;
}

function normalizeSlug(str: string): string {
  if (!str) return "";
  const decoded = safeDecode(str);
  return decoded
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

function getBaseProductName(fullName: string): string {
  if (!fullName) return "";
  return fullName
    .replace(/\s*[\(\[\{]?(XS|S|M|L|XL|Única|Unica)[\]\}\)]?\s*$/i, "")
    .replace(/\s*-\s*(XS|S|M|L|XL|Única|Unica)$/i, "")
    .trim();
}

function parsePrice(priceVal: unknown): number {
  if (typeof priceVal === "number") return priceVal;
  if (!priceVal) return 0;
  const cleaned = String(priceVal)
    .replace(/[^0-9.,]/g, "")
    .replace(",", ".");
  return parseFloat(cleaned) || 0;
}

"use client";

import React, { createContext, useContext, useState } from "react";

export interface CartItem {
  id: string | number;
  name: string;
  price: string | number;
  size: string;
  quantity: number;
  image?: string;
}

export interface CartContextType {
  cart: CartItem[];
  isCartOpen: boolean;
  openCart: () => void;
  closeCart: () => void;
  addToCart: (item: CartItem) => void;
  removeFromCart: (id: string | number, size?: string) => void;
  updateQuantity: (
    id: string | number,
    size: string | undefined,
    quantity: number,
  ) => void;
  clearCart: () => void;
}

const CartContext = createContext<CartContextType | undefined>(undefined);

export function CartProvider({ children }: { children: React.ReactNode }) {
  const [cart, setCart] = useState<CartItem[]>([]);
  const [isCartOpen, setIsCartOpen] = useState<boolean>(false);

  const openCart = () => setIsCartOpen(true);
  const closeCart = () => setIsCartOpen(false);

  const addToCart = (newItem: CartItem) => {
    setCart((prevCart) => {
      const existingIndex = prevCart.findIndex(
        (item) => item.id === newItem.id && item.size === newItem.size,
      );
      if (existingIndex > -1) {
        const updated = [...prevCart];
        updated[existingIndex].quantity += newItem.quantity || 1;
        return updated;
      }
      return [...prevCart, { ...newItem, quantity: newItem.quantity || 1 }];
    });
    openCart();
  };

  const removeFromCart = (id: string | number, size?: string) => {
    setCart((prevCart) =>
      prevCart.filter((item) => !(item.id === id && item.size === size)),
    );
  };

  const updateQuantity = (
    id: string | number,
    size: string | undefined,
    quantity: number,
  ) => {
    if (quantity <= 0) {
      removeFromCart(id, size);
      return;
    }
    setCart((prevCart) =>
      prevCart.map((item) =>
        item.id === id && item.size === size ? { ...item, quantity } : item,
      ),
    );
  };

  const clearCart = () => setCart([]);

  return (
    <CartContext.Provider
      value={{
        cart,
        isCartOpen,
        openCart,
        closeCart,
        addToCart,

// 1. Decodificación segura de URLs
function safeDecode(str: string): string {
  if (!str) return "";
  let decoded = str;
  try {
    for (let i = 0; i < 3; i++) {
      if (decoded.includes("%")) {
        const temp = decodeURIComponent(decoded);
        if (temp === decoded) break;
        decoded = temp;
      }
    }
  } catch (e) {
    console.error("Error decodificando el slug:", e);
  }
  return decoded;
}

// 2. Normalización de cadenas para comparaciones
function normalizeSlug(str: string): string {
  if (!str) return "";
  const decoded = safeDecode(str);
  return decoded
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

// 3. Extraer el nombre base quitando la talla y color al final
function getBaseProductName(fullName: string): string {
  if (!fullName) return "";
  return fullName
    .replace(/\s*[\(\[\{]?(XS|S|M|L|XL|Única|Unica)[\]\}\)]?\s*$/i, "")
    .replace(/\s*-\s*(XS|S|M|L|XL|Única|Unica)$/i, "")
    .trim();
}

// 4. Convertir precios a número puro
function parsePrice(priceVal: any): number {
  if (typeof priceVal === "number") return priceVal;
  if (!priceVal) return 0;
  const cleaned = String(priceVal)
    .replace(/[^0-9.,]/g, "")
    .replace(",", ".");
  return parseFloat(cleaned) || 0;
}

"use client";

import React, {
  createContext,
  useContext,
  useEffect,
  useState,
  type ReactNode,
} from "react";
import { type Order } from "@/lib/orders";

interface AdminOrdersContextType {
  orders: Order[];
  setOrders: React.Dispatch<React.SetStateAction<Order[]>>;
}

const AdminOrdersContext = createContext<AdminOrdersContextType | undefined>(
  undefined,
);

export function AdminOrdersProvider({ children }: { children: ReactNode }) {
  const [orders, setOrders] = useState<Order[]>([]);

  useEffect(() => {
    async function loadOrders() {
      try {
        const response = await fetch("/api/orders");

        if (!response.ok) {
          // 1. Leemos el cuerpo como texto plano UNA SOLA VEZ
          const rawText = await response.text();
          let detalleError = rawText;

          // 2. Intentamos parsear el texto a JSON si aplica
          try {
            const errorData = JSON.parse(rawText);
            detalleError =
              errorData.message || errorData.error || JSON.stringify(errorData);
          } catch {
            // Si no es JSON (es HTML o texto plano), conserva el rawText
          }

          // 3. Imprimimos el mensaje formateado como string para ver el detalle real en pantalla
          console.error(
            `[API Error ${response.status} ${response.statusText}]: ${detalleError}`,
          );

          throw new Error(
            `Error HTTP ${response.status}: ${response.statusText || "Fallo en la API"}`,
          );
        }

        const data = await response.json();

        if (data.success && Array.isArray(data.orders)) {
          setOrders(data.orders);
        } else if (Array.isArray(data.orders)) {
          setOrders(data.orders);
        } else if (Array.isArray(data)) {
          setOrders(data);
        }
      } catch (error) {
        console.error("Error cargando pedidos:", error);
      }
    }

    loadOrders();
  }, []);

  return (
    <AdminOrdersContext.Provider
      value={{
        orders,
        setOrders,
      }}
    >
      {children}
    </AdminOrdersContext.Provider>
  );
}

export function useAdminOrders() {
  const context = useContext(AdminOrdersContext);

  if (!context) {
    throw new Error("useAdminOrders must be used inside AdminOrdersProvider");
  }

  return context;
}

export interface InventoryVariant {
  id: string;
  color: string;
  size: string;
  sku: string;
  initialStock: number;
  stock: number;
}

export interface InventoryItem {
  id: string;
  name: string;
  category: string;

  // Precio que tú estipulas para pagos directos en $
  dollarPrice: number;

  // Precio publicado calculado con BNC / BCV
  price: number;

  cost: number;

  webVisible: boolean;

  images: string[];

  sizeGuideImages: string[];

  careDetails: string;

  variants: InventoryVariant[];
}

export interface Rates {
  tasaBCV: number;
  tasaBNC: number;
  discountPercent?: number;
}

export const DEFAULT_RATES: Rates = {
  tasaBCV: 745,
  tasaBNC: 1000,
  discountPercent: 15,
};

/*
  Calcula el precio publicado.

  Precio estipulado en $
  × BNC
  ÷ BCV

  Si roundPricesUp está activado,
  se redondea hacia arriba.
*/
export function calculatePricing(
  dollarPrice: number,
  rates: Rates,
  roundPricesUp = true,
) {
  const tasaBCV =
    rates.tasaBCV && rates.tasaBCV > 0 ? rates.tasaBCV : DEFAULT_RATES.tasaBCV;

  const tasaBNC =
    rates.tasaBNC && rates.tasaBNC > 0 ? rates.tasaBNC : DEFAULT_RATES.tasaBNC;

  const discountPercent =
    typeof rates.discountPercent === "number"
      ? rates.discountPercent
      : DEFAULT_RATES.discountPercent;

  if (!dollarPrice || dollarPrice <= 0) {
    return {
      priceBsFormatted: "0,00",
      priceBCVFormatted: "0.00",
      cashUSDFormatted: "0.00",
      discountPercent,
    };
  }

  // Precio publicado antes del redondeo
  const rawPublishedUSD = (dollarPrice * tasaBNC) / tasaBCV;

  // Precio publicado final
  const publishedUSD = roundPricesUp
    ? Math.ceil(rawPublishedUSD)
    : rawPublishedUSD;

  // Los Bs se calculan usando el precio publicado FINAL
  const publishedBs = publishedUSD * tasaBCV;

  return {
    priceBsFormatted: publishedBs.toLocaleString("es-VE", {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    }),

    priceBCVFormatted: publishedUSD.toFixed(2),

    cashUSDFormatted: dollarPrice.toFixed(2),

    discountPercent,
  };
}

import { createServerClient } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createClient() {
  const cookieStore = await cookies();

  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll();
        },
        setAll(cookiesToSet) {
          try {
            cookiesToSet.forEach(({ name, value, options }) =>
              cookieStore.set(name, value, options),
            );
          } catch {
            // Puede ocurrir cuando las cookies
            // se modifican desde un Server Component.
          }
        },
      },
    },
  );
}
